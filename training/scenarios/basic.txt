# 기본 시연 시나리오 (명령 인식률 측정용) — 한 줄에 하나 또는 쉼표로 구분
# 사용 가능한 명령: click, swipe_left, swipe_right, confirm, back, home
click          # 대기화면 '주문 시작'
click          # 홈 → '버거'
swipe_left     # 다음 페이지
swipe_right    # 이전 페이지
click          # 불고기버거 → 상세
click          # '세트 담기' (담고 목록으로)
home           # ✌️ 처음(홈)
click          # '음료'
click          # 콜라 → 상세
confirm        # 👍 담기
click          # '장바구니 보기'
click          # ＋ 수량 1개 추가
confirm        # 👍 주문하기 → 주문번호
confirm        # 👍 팝업 닫기 → 대기 화면
